Google Sheets Database Sync

Connects directly to your `Teams`, `Schedule`, `Results`, and `PlanRules` sheets.

Google Apps Script Backend (`Code.gs`)

Create tabs Teams, Schedule, Results, and PlanRules in your Google Sheet, open Extensions → Apps Script, paste this code, and deploy as a Web App (Execute as: Me, Who has access: Anyone): 

const SHEET_NAMES = ['Teams', 'Schedule', 'Results', 'PlanRules'];
const ADMIN_PASSCODE = 'DIG2026';

function ensureSheets_(ss) {
  SHEET_NAMES.forEach((name) => {
    if (!ss.getSheetByName(name)) ss.insertSheet(name);
  });
}

function sheetToObjects_(sheet) {
  const values = sheet.getDataRange().getValues();
  if (values.length < 2) return [];
  const headers = values[0].map((h) => String(h).trim());
  return values.slice(1)
    .filter((row) => row.some((cell) => cell !== ''))
    .map((row) => {
      const obj = {};
      headers.forEach((h, idx) => { if (h) obj[h] = row[idx]; });
      return obj;
    });
}

function writeObjectsToSheet_(sheet, rows) {
  sheet.clearContents();
  if (!Array.isArray(rows) || rows.length === 0) return;
  const headers = Object.keys(rows[0]);
  const output = [headers];
  rows.forEach((item) => {
    output.push(headers.map((h) => (item[h] !== undefined && item[h] !== null ? item[h] : '')));
  });
  sheet.getRange(1, 1, output.length, headers.length).setValues(output);
}

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  ensureSheets_(ss);
  const requestedTab = e && e.parameter && e.parameter.sheet ? String(e.parameter.sheet) : 'ALL';
  const payload = {};
  if (requestedTab === 'ALL') {
    SHEET_NAMES.forEach((name) => { payload[name] = sheetToObjects_(ss.getSheetByName(name)); });
  } else if (SHEET_NAMES.includes(requestedTab)) {
    payload[requestedTab] = sheetToObjects_(ss.getSheetByName(requestedTab));
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true, data: payload }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  ensureSheets_(ss);
  const body = e && e.postData && e.postData.contents ? JSON.parse(e.postData.contents) : {};
  if (body.passcode !== ADMIN_PASSCODE) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'Unauthorized passcode' }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  const tables = body.tables || {};
  SHEET_NAMES.forEach((name) => {
    if (Array.isArray(tables[name])) {
      writeObjectsToSheet_(ss.getSheetByName(name), tables[name]);
    }
  });
  return ContentService.createTextOutput(JSON.stringify({ ok: true, savedAt: new Date().toISOString() }))
    .setMimeType(ContentService.MimeType.JSON);
}
